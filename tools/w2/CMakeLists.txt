cmake_minimum_required(VERSION 3.16)
project(delabella_loop CXX)
find_package(OpenCASCADE REQUIRED)
add_executable(delabella_loop delabella_loop.cpp)
if(NOT OpenCASCADE_INCLUDE_DIR AND OpenCASCADE_INCLUDE_DIRS)
  set(OpenCASCADE_INCLUDE_DIR "${OpenCASCADE_INCLUDE_DIRS}")
endif()
target_include_directories(delabella_loop PRIVATE ${OpenCASCADE_INCLUDE_DIR} ${OpenCASCADE_INCLUDE_DIRS})
target_link_libraries(delabella_loop PRIVATE ${OpenCASCADE_LIBRARIES})
